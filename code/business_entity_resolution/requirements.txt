# Dependencies for the business entity resolution pipeline.
#
#     pip install -r requirements.txt
#
# These versions are the ones the pipeline was developed and tested against, and
# are the versions resolved in the working environment. They are *not* a lockfile
# for a different platform: pip may need to build wheels for your interpreter, and
# the floor matters more than the exact patch level (see the notes per package).
# Re-pin from what actually installs before relying on this elsewhere.

# --- core data handling -----------------------------------------------------
# NumPy 2 is required, not merely allowed: the caches store fixed-width byte
# arrays and the pipeline relies on BytesDType plus np.lib.format.open_memmap.
numpy==2.5.0
pandas==3.0.5

# --- on-disk columnar cache -------------------------------------------------
# The normalised records are cached as Parquet rather than pickled .npy so that
# a half-written cache is a visible, recoverable artefact instead of a corrupt
# file that unpickles into garbage. pyarrow is the engine for both writing and
# the row-count/footer checks that validate a cache.
pyarrow==25.0.1

# --- similarity / string metrics -------------------------------------------
# rapidfuzz provides C-accelerated Levenshtein, Jaro-Winkler and token-set ratios.
# Edit distance dominates feature cost once candidate volume grows, so the
# pure-Python alternatives (difflib) are not fast enough.
# Note: JaroWinkler moved to rapidfuzz.distance in 3.x; importing it from
# rapidfuzz.fuzz raises ImportError on this version.
rapidfuzz==3.14.6
# jellyfish covers the phonetic encodings (metaphone/soundex) that rapidfuzz does
# not, used by the name_phonetic similarity feature. The corresponding *blocking*
# strategy was measured to contribute nothing and is disabled, but the feature
# itself is still computed.
jellyfish==1.2.1

# --- unicode handling -------------------------------------------------------
# Transliterates Devanagari (and other non-Latin) names and addresses so string
# similarity works across scripts. Pure Python, no external data.
Unidecode==1.4.0

# --- modelling --------------------------------------------------------------
# Gradient-boosted trees over the handcrafted features. Trains in seconds on a
# pair sample and yields the calibrated per-candidate probability that threshold
# tuning needs. Exactly one of lightgbm / xgboost should be installed; model.py
# imports lightgbm.
lightgbm==4.7.0
